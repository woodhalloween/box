```python
def validate_analysis_output(csv_file):
    """出力ファイルの妥当性をチェック"""
    df = pd.read_csv(csv_file)
    
    # 1. フレーム数チェック
    if len(df) < 100:
        logger.warning(f"⚠️ フレーム数が少ない: {len(df)}")
    
    # 2. データ有効性チェック
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    non_zero_cols = [col for col in numeric_cols 
                     if (df[col] != 0).any() 
                     and col != 'frame_number']
    
    if len(non_zero_cols) < len(numeric_cols) * 0.5:
        logger.warning(f"⚠️ 検出されたデータが少ない: {len(non_zero_cols)}/{len(numeric_cols)}")
    
    # 3. ポーズ推定の有効性チェック
    visibility_cols = [col for col in df.columns if 'visibility' in col]
    avg_visibility = sum(df[col].mean() for col in visibility_cols) / len(visibility_cols)
    
    if avg_visibility < 0.5:
        logger.warning(f"⚠️ ポーズ推定の信頼度が低い: {avg_visibility:.1%}")
    
    return len(df) >= 100 and avg_visibility >= 0.5
```
